function WhotShapeMark({ shape, size = "large" }: { shape: WhotCard["shape"]; size?: "small" | "large" }) {
  if (shape === "whot") return <span className={`whot-shape-mark is-whot is-${size}`} aria-hidden="true">W</span>;
  return <span className={`whot-shape-mark shape-${shape} is-${size}`} aria-hidden="true" />;
}

interface SeatPoint { left: number; top: number; }

// WHOT uses deliberate table seats rather than a generic semicircle. The seven-
// opponent layout matches the approved 3.5.7 mock-up: four seats across the far
// rail, two down the left rail and one on the right, leaving the centre and local
// hand visually clear. Smaller tables use balanced subsets of the same geometry.
const WHOT_SEAT_PRESETS: Record<number, SeatPoint[]> = {
  1: [{ left: 50, top: 10 }],
  2: [{ left: 36, top: 10 }, { left: 64, top: 10 }],
  3: [{ left: 25, top: 11 }, { left: 50, top: 8 }, { left: 75, top: 11 }],
  4: [{ left: 21, top: 11 }, { left: 40, top: 8 }, { left: 60, top: 8 }, { left: 79, top: 11 }],
  5: [{ left: 24, top: 10 }, { left: 50, top: 8 }, { left: 76, top: 10 }, { left: 8, top: 43 }, { left: 92, top: 43 }],
  6: [{ left: 21, top: 10 }, { left: 40, top: 8 }, { left: 60, top: 8 }, { left: 79, top: 10 }, { left: 8, top: 44 }, { left: 92, top: 44 }],
  7: [{ left: 21, top: 10 }, { left: 40, top: 8 }, { left: 60, top: 8 }, { left: 79, top: 10 }, { left: 8, top: 37 }, { left: 92, top: 45 }, { left: 8, top: 59 }],
};

function opponentSeat(index: number, total: number): SeatPoint {
  const safeTotal = Math.max(1, Math.min(7, total));
  return WHOT_SEAT_PRESETS[safeTotal]?.[Math.max(0, Math.min(index, safeTotal - 1))] ?? { left: 50, top: 10 };
}
function opponentCardTarget(index: number, total: number): SeatPoint {
  return opponentSeat(index, total);
}

function WhotCardView({ card, playable, onClick, fanIndex, fanTotal, topCard = false }: { card: WhotCard; playable?: boolean; onClick?: () => void; fanIndex?: number; fanTotal?: number; topCard?: boolean }) {
  const centredIndex = fanIndex !== undefined && fanTotal !== undefined ? fanIndex - (fanTotal - 1) / 2 : 0;
  const total = fanTotal ?? 1;
  const angleStep = total <= 5 ? 4 : total <= 8 ? 3 : total <= 12 ? 2 : 1.25;
  const overlap = total <= 5 ? 7 : total <= 8 ? -2 : total <= 12 ? -12 : total <= 16 ? -22 : -30;
  const style = fanIndex !== undefined && fanTotal !== undefined ? {
    "--whot-fan-angle": `${Math.max(-16, Math.min(16, centredIndex * angleStep))}deg`,
    "--whot-fan-rise": `${Math.abs(centredIndex) * Math.min(2.6, 13 / Math.max(1, total - 1))}px`,
    "--whot-card-order": fanIndex + 1,
    marginLeft: fanIndex === 0 ? 0 : `${overlap}px`,
  } as CSSProperties : undefined;
  const label = card.shape === "whot" ? "WHOT 20" : `${card.number} ${card.shape}`;
  return <button type="button" aria-label={label} style={style} className={`whot-card whot-card-authentic shape-${card.shape} ${topCard ? "is-top-card" : ""} ${card.shape === "whot" ? "is-whot" : ""} ${playable ? "is-playable" : ""}`} disabled={!playable} onClick={onClick}>
    <span className="whot-card-corner is-top"><b>{card.number}</b><WhotShapeMark shape={card.shape} size="small" /></span>
    <span className="whot-card-centre" aria-hidden="true">
      {card.shape === "whot" ? <><em>WHOT</em><small>CALL SHAPE</small></> : <WhotShapeMark shape={card.shape} />}
      {card.shape === "star" && <i className="whot-star-score">×2</i>}
    </span>
    <span className="whot-card-corner is-bottom"><b>{card.number}</b><WhotShapeMark shape={card.shape} size="small" /></span>
  </button>;
}

function WhotMotionLayer({ state, viewerId }: { state: WhotPublicState; viewerId: string | null }) {
  const motion = state.motion;
  if (!motion) return null;
  const viewer = state.players.find((player) => player.id === viewerId);
  const opponents = state.players.filter((player) => player.id !== viewer?.id);
  const positionForPlayer = (playerId: string | null): SeatPoint => {
    if (playerId && viewer?.id === playerId) return { left: 50, top: 88 };
    const index = opponents.findIndex((player) => player.id === playerId);
    if (index >= 0) return opponentCardTarget(index, opponents.length);
    if (!viewer) {
      const allIndex = state.players.findIndex((player) => player.id === playerId);
      const angle = (Math.PI * 2 * Math.max(0, allIndex)) / Math.max(1, state.players.length) - Math.PI / 2;
      return { left: 50 + Math.cos(angle) * 37, top: 50 + Math.sin(angle) * 36 };
    }
    return { left: 50, top: 12 };
  };

  const cards: Array<{ key: string; face: boolean; from: SeatPoint; to: SeatPoint; delay: number }> = [];
  const centre = { left: 50, top: 47 };
  if (motion.kind === "play") {
    cards.push({ key: "play", face: true, from: positionForPlayer(motion.playerId), to: centre, delay: 0 });
  } else if (motion.kind === "draw") {
    const target = positionForPlayer(motion.playerId);
    for (let index = 0; index < Math.min(16, Math.max(1, motion.count)); index += 1) cards.push({ key: `draw-${index}`, face: false, from: centre, to: target, delay: index * 145 });
  } else if (motion.kind === "general-market") {
    cards.push({ key: "market-play", face: true, from: positionForPlayer(motion.playerId), to: centre, delay: 0 });
    motion.targetPlayerIds.forEach((playerId, index) => cards.push({ key: `market-${playerId}-${index}`, face: false, from: centre, to: positionForPlayer(playerId), delay: 520 + index * 180 }));
  } else {
    const targets = motion.targetPlayerIds.length ? motion.targetPlayerIds : state.players.map((player) => player.id);
    const totalCards = Math.min(36, Math.max(targets.length, motion.count));
    for (let index = 0; index < totalCards; index += 1) {
      const playerId = targets[index % targets.length];
      cards.push({ key: `deal-${index}`, face: false, from: centre, to: positionForPlayer(playerId), delay: index * 72 });
    }
  }

  return <div key={motion.sequence} className={`whot-motion-layer motion-${motion.kind}`} aria-hidden="true">
    {cards.map((entry, index) => {
      const style = {
        "--motion-from-left": `${entry.from.left}%`, "--motion-from-top": `${entry.from.top}%`, "--motion-to-left": `${entry.to.left}%`, "--motion-to-top": `${entry.to.top}%`, "--motion-delay": `${entry.delay}ms`, "--motion-index": index,
      } as CSSProperties;
      return entry.face && motion.card ? <div key={entry.key} style={style} className={`whot-motion-card is-face shape-${motion.card.shape}`}><span className="whot-motion-corner">{motion.card.number}</span><WhotShapeMark shape={motion.card.shape} /></div> : <div key={entry.key} style={style} className="whot-motion-card is-back"><span className="whot-back-frame"><strong>W</strong><small>HALIEUS WHOT</small></span></div>;
    })}
  </div>;
}

          {viewer && <div className="whot357-you"><span>{viewer.name.slice(0,1).toUpperCase()}</span><div><small>PLAYER</small><strong>{viewer.name}</strong><em>{state.isSpectator ? "Spectating" : "You"}</em></div></div>}
        </aside>
      </section>}
    </section>}
    {shapeCardId && <div className="modal-backdrop game-menu-top-layer" role="presentation"><section className="game-menu-modal"><div className="modal-handle"/><p className="modal-eyebrow">WHOT 20</p><h2>Call a shape</h2><div className="whot-shape-grid">{SHAPES.map((shape) => <button type="button" key={shape} onClick={() => { onPlay(shapeCardId, shape); setShapeCardId(null); }}><strong>{shapeGlyph[shape]}</strong><span>{shape}</span></button>)}</div><button type="button" className="button-ghost full-button" onClick={() => setShapeCardId(null)}>Cancel</button></section></div>}
    {inviteOpen && <div className="modal-backdrop" role="presentation" onMouseDown={(event) => event.target === event.currentTarget && setInviteOpen(false)}><section className="card-game-invite-modal"><button type="button" className="icon-button invite-modal-close" onClick={() => setInviteOpen(false)}>×</button><InviteLobbyPanel roomCode={state.code} darkMode={darkMode} gameTitle="WHOT" invitePathPrefix="/whot" accent={WHOT_ACCENT} theme={theme} /></section></div>}
    {menuOpen && <div className="modal-backdrop game-menu-top-layer" role="presentation" onMouseDown={(event) => event.target === event.currentTarget && setMenuOpen(false)}><section className="game-menu-modal card-game-menu-modal"><div className="modal-handle"/><header className="modal-heading-row"><div><p className="modal-eyebrow">WHOT · {modeLabel} · Room {state.code}</p><h2>Game menu</h2></div><button type="button" className="icon-button" onClick={() => setMenuOpen(false)}>×</button></header><button type="button" className="menu-action menu-action-primary" onClick={() => setMenuOpen(false)}><span>▶</span><span><strong>Resume game</strong><small>Return to the table</small></span></button><DisplaySettingsPanel darkMode={darkMode} soundEnabled={soundEnabled} onToggleDarkMode={onToggleDarkMode} onToggleSound={onToggleSound} onToggleFullscreen={onToggleFullscreen}/><button type="button" className="menu-action" onClick={() => downloadWhotGameReport(state)}><span>📄</span><span><strong>Download Game Report</strong><small>Export table state, action history and AI evidence</small></span></button><RecoveryKeyPanel recoveryKey={recoveryKey} /><button type="button" className="menu-action" onClick={onLeave}><span>←</span><span><strong>Back to Game Room</strong><small>Keep your seat recoverable</small></span></button>{viewer && !state.isSpectator && <button type="button" className="menu-action menu-action-danger" onClick={() => { setMenuOpen(false); setConfirmAction("forfeit"); }}><span>⚑</span><span><strong>Forfeit seat</strong><small>Leave this WHOT game permanently</small></span></button>}{isHost && <button type="button" className="menu-action menu-action-danger" onClick={() => { setMenuOpen(false); setConfirmAction("end"); }}><span>■</span><span><strong>End WHOT game</strong><small>Finalise and archive this session for everyone</small></span></button>}</section></div>}
    <ConfirmDialog open={confirmAction !== null} title={confirmAction === "end" ? "End WHOT game?" : "Forfeit your WHOT seat?"} message={confirmAction === "end" ? "This will finalise the WHOT room for everyone and archive the session." : "Your seat will close permanently and the game will continue without you."} confirmLabel={confirmAction === "end" ? "End game" : "Forfeit"} destructive onCancel={() => setConfirmAction(null)} onConfirm={() => { const action = confirmAction; setConfirmAction(null); if (action === "end") onEndGame(); else if (action === "forfeit") onForfeit(); }} />
  </main>;
}
